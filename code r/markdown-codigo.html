# Función de Cp de Mallows basada en la guía del docente
Cp_calc <- function(mod_cand, mod_comp) {
  n <- nrow(data3)
  p <- length(coef(mod_cand))
  s2_full <- summary(mod_comp)$sigma^2
  
  deviance(mod_cand) / s2_full - n + 2 * p
}

  R2_aj = c(summary(mod_forward)$adj.r.squared,
            summary(mod_backward)$adj.r.squared,
            summary(mod_stepwise)$adj.r.squared),
  Cp = c(Cp_calc(mod_forward, mod_completo),
         Cp_calc(mod_backward, mod_completo),
         Cp_calc(mod_stepwise, mod_completo)),
  AIC = c(AIC(mod_forward),
          AIC(mod_backward),
          AIC(mod_stepwise)),
  BIC = c(BIC(mod_forward),
          BIC(mod_backward),
          BIC(mod_stepwise))
)
